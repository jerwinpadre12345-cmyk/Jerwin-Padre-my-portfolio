My Java Programming Project

Number Guessing Game

This is my Java programming project for CC102. The game generates a random number from 1 to 100. The player needs to guess the correct number. It uses loops, conditions, variables, Scanner, and Random.


Java Source Code


import java.util.Scanner;
import java.util.Random;

public class Main {
    public static void main(String[] args) {

        Scanner input = new Scanner(System.in);
        Random random = new Random();

        int score = 0;
        int playAgain = 1;

        System.out.println("==============================");
        System.out.println("    NUMBER GUESSING GAME");
        System.out.println("==============================");

        while (playAgain == 1) {

            int secretNumber = random.nextInt(100) + 1;
            int guess = 0;
            int attempts = 0;

            System.out.println("\nGuess a number from 1 to 100!");

            while (guess != secretNumber) {

                System.out.print("Enter your guess: ");
                guess = input.nextInt();
                attempts++;

                if (guess < 1 || guess > 100) {
                    System.out.println("Enter numbers from 1 to 100 only!");
                }

                else if (guess < secretNumber) {
                    System.out.println("Too low! Try again.");
                }

                else if (guess > secretNumber) {
                    System.out.println("Too high! Try again.");
                }

                else {
                    System.out.println("\nCongratulations!");
                    System.out.println("Correct number!");
                    System.out.println("Attempts: " + attempts);

                    score += 100 - (attempts * 5);

                    if (score < 0) {
                        score = 0;
                    }

                    System.out.println("Your score: " + score);
                }
            }

            System.out.println("\nPlay again?");
            System.out.println("1 - Yes");
            System.out.println("0 - No");
            System.out.print("Enter choice: ");

            playAgain = input.nextInt();
        }

        System.out.println("\n==============================");
        System.out.println("          GAME OVER");
        System.out.println("Final Score: " + score);
        System.out.println("==============================");

        input.close();
    }
}
